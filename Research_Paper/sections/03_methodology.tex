\section{Methodology}
\label{sec:methodology}

\subsection{Subjective Logic Dirichlet Frame}
Let $\mathbb{X} = \{\theta_1, \theta_2, \theta_3, \theta_4\}$ denote the 4-state flood risk frame corresponding to $\text{0: Dry}, \text{1: Saturated}, \text{2: SurfaceFlow}, \text{3: Inundation}$.
An evidential opinion emitted by specialist agent $A_i$ is defined as $\omega_{A_i} = (\mathbf{b}_{A_i}, u_{A_i}, \mathbf{a}_{A_i})$, satisfying:
\begin{equation}
\sum_{k=1}^4 b_k + u = 1.0, \quad b_k \ge 0, \quad u \ge 0, \quad \sum_{k=1}^4 a_k = 1.0
\end{equation}

Given Dirichlet evidence parameters $\boldsymbol{\alpha} = (\alpha_1, \alpha_2, \alpha_3, \alpha_4)$ with total evidence mass $C = \sum_{k=1}^4 \alpha_k$, the bijection maps:
\begin{equation}
b_k = \frac{\alpha_k - C \cdot a_k}{C + \sum r_k}, \qquad u = \frac{C}{C + \sum r_k}
\end{equation}
Expected (projected) probability for state $\theta_k$ is:
\begin{equation}
\mathbf{P}(\theta_k) = b_k + a_k \cdot u
\end{equation}

\subsection{Partial-Observable Projection (Kaplan et al., 2015)}
When sensory input for agent $A_i$ is missing (e.g., cloud occlusion or sensor dropout), the evidence mass reverts to the prior:
\begin{equation}
\alpha_k^{\text{proj}} = C \cdot a_k \implies u \to 1.0, \quad b_k \to 0.0
\end{equation}
This mathematically guarantees that missing inputs decay into epistemic uncertainty $u$ monotonically.

\subsection{Deterministic Orchestrator & Fusion Selection}
The SL Orchestrator evaluates pairwise Jensen-Shannon (JS) divergence across agent opinions:
\begin{equation}
\text{JS}(p, q) = \frac{1}{2} D_{KL}(p \parallel M) + \frac{1}{2} D_{KL}(q \parallel M)
\end{equation}
- **High Agreement ($\text{JS} < \tau_{\text{low}} = 0.1$)**: Route to **Consensus \& Compromise Fusion (CCF)**.
- **Conflict ($\text{JS} \ge \tau_{\text{low}}$)**: Route to **Weighted Belief Fusion (WBF)** weighted by agent reputation $\gamma_i$.

\subsection{Dynamic Credibility Tracking (Wang \& Singh, 2007)}
Agent reputation $\gamma_i \in [\gamma_{\text{min}}, 1.0]$ updates dynamically using Brier score accuracy verification against historical ground truth:
\begin{equation}
\text{BS}_i = \frac{1}{4} \sum_{k=1}^4 (\hat{p}_{i,k} - y_k)^2 \implies \gamma_i^{(t+1)} = \text{clip}\left(\gamma_i^{(t)} + \eta (1 - 2 \text{BS}_i), \gamma_{\text{min}}, 1.0\right)
\end{equation}

\subsection{Hybrid Evidential Head}
The prediction head receives a 27-dimensional feature vector concatenating the 11-dimensional fused SL opinion tensor $(\mathbf{b}, u, \text{max\_js}, \boldsymbol{\gamma})$ with the 16-dimensional raw feature vector ($\text{ERA5}, \text{SAR}, \text{DEM}, \text{OpenAQ}$). A LightGBM multi-class classifier predicts discrete flood state while a LightGBM regressor estimates continuous water depth $d \ge 0$.
