\label{app:algorithm}
Table~\ref{tab:coordinator_full} provides the detailed implementation of the Subjective Logic Orchestrator.

\begin{table}[htbp]
\caption{Full AEGIS-SL Evidence Routing and Opinion Fusion}
\label{tab:coordinator_full}
\centering
\resizebox{\columnwidth}{!}{%
\begin{tabular}{ll}
\toprule
\multicolumn{2}{l}{\textbf{Full AEGIS-SL Evidence Routing and Opinion Fusion Algorithm}} \\
\midrule
\textbf{Require:} & Cell ID $c$, Date $t$, Context $X$, Agents $\mathcal{A}$, $\tau_{\text{low}}=0.1$, $\eta=0.05$ \\
\textbf{Ensure:} & Fused Opinion $\omega_{\text{fused}}$, Projected Probabilities $\mathbf{P}$, Composite Vector $\mathbf{F} \in \mathbb{R}^{27}$ \\
\midrule
1: & \textbf{for each} agent $A_i \in \mathcal{A}$ \textbf{do} \\
2: & \quad \textbf{if} $X$ contains valid modality observations for $A_i$ \textbf{then} \\
3: & \quad \quad $\omega_{A_i} \leftarrow A_i.\text{emit\_opinion}(c, t, X)$ \\
4: & \quad \textbf{else} \\
5: & \quad \quad $\omega_{A_i} \leftarrow \text{PartialObservableProjection}(\mathbf{a}_{A_i})$ \\
6: & \quad \textbf{end if} \\
7: & \textbf{end for} \\
8: & $\text{JS}_{\text{max}} \leftarrow \max_{i < j} \text{JS}(\mathbf{P}_{A_i}, \mathbf{P}_{A_j})$ \\
9: & \textbf{if} $\text{JS}_{\text{max}} < \tau_{\text{low}}$ \textbf{then} \\
10: & \quad $\omega_{\text{fused}} \leftarrow \text{ConsensusCompromiseFusion}(\{\omega_{A_i}\})$ \\
11: & \textbf{else} \\
12: & \quad $\omega_{\text{fused}} \leftarrow \text{WeightedBeliefFusion}(\{\omega_{A_i}\}, \{\gamma_{A_i}\})$ \\
13: & \textbf{end if} \\
14: & $\mathbf{P}(\theta_k) \leftarrow b_k^{\text{fused}} + a_k \cdot u^{\text{fused}} \quad \forall k \in \{1..4\}$ \\
15: & $\mathbf{F} \leftarrow [b_1, b_2, b_3, b_4, u, \text{JS}_{\text{max}}, \gamma_1, \dots, \gamma_5, X_{\text{raw}}]$ \\
16: & \textbf{return} $\omega_{\text{fused}}, \mathbf{P}, \mathbf{F}$ \\
\bottomrule
\end{tabular}%
}
\end{table}

CREATE TABLE IF NOT EXISTS spatial_grid (
    cell_id INTEGER PRIMARY KEY,
    lat DOUBLE, lon DOUBLE,
    land_cover INTEGER, slope_deg DOUBLE,
    impervious_frac DOUBLE, elevation_m DOUBLE
);

CREATE TABLE IF NOT EXISTS agent_opinions_log (
    log_id VARCHAR PRIMARY KEY,
    cell_id INTEGER, target_date DATE,
    b0 DOUBLE, b1 DOUBLE, b2 DOUBLE, b3 DOUBLE,
    u DOUBLE, max_js DOUBLE, operator VARCHAR,
    created_at TIMESTAMP
);
\end{verbatim}

\section{Per-Agent Credibility ($\gamma$) Time-Series Progression}
\label{app:credibility_progression}
Over the 24 daily steps of the Brisbane 2022 event, per-agent credibility $\gamma_i$ evolves dynamically:
\begin{itemize}
    \item \texttt{ClimateAgent} ($\gamma_1$): Starts at $1.0$, stabilizes at $0.95$ due to steady ERA5 precipitation coverage.
    \item \texttt{SatelliteAgent} ($\gamma_2$): Starts at $1.0$, experiences brief drop to $0.82$ during orbit gap, recovers to $0.94$.
    \item \texttt{LandCoverAgent} ($\gamma_3$): Maintains constant $\gamma_3 = 0.98$ due to static physical terrain attributes.
    \item \texttt{AirQualityAgent} ($\gamma_4$): Fluctuates between $0.85$ and $0.92$ during sensor washout periods.
    \item \texttt{DocIntAgent} ($\gamma_5$): Drops to $\gamma_5 = 0.70$ during dates with no published emergency bulletins (vacuous prior).
\end{itemize}
