\section{Mathematical Foundations}
\label{sec:methodology}

\subsection{Opinion Algebra and Dirichlet Bijection}
Let $\mathbb{X} = \{\theta_1, \theta_2, \theta_3, \theta_4\}$ denote a discrete, exhaustive frame of discernment corresponding to four urban flood risk states:
$$\theta_1 = \text{Dry}, \quad \theta_2 = \text{Saturated}, \quad \theta_3 = \text{SurfaceFlow}, \quad \theta_4 = \text{Inundation}$$

An evidential opinion emitted by a specialist agent $A_i$ over frame $\mathbb{X}$ is represented as a tuple:
\begin{equation}
\omega_{A_i} = \left( \mathbf{b}_{A_i}, u_{A_i}, \mathbf{a}_{A_i} \right)
\end{equation}
where $\mathbf{b}_{A_i} = (b_1, b_2, b_3, b_4)^T$ is the belief mass vector, $u_{A_i} \ge 0$ is the scalar epistemic uncertainty, and $\mathbf{a}_{A_i} = (a_1, a_2, a_3, a_4)^T$ is the non-informative base-rate prior. The opinion components satisfy the core Subjective Logic conservation law \cite{josang2016subjective}:
\begin{equation}
\sum_{k=1}^4 b_k + u = 1.0, \quad b_k \ge 0, \quad u \ge 0, \quad \sum_{k=1}^4 a_k = 1.0
\label{eq:sl_conservation}
\end{equation}

Given Dirichlet evidence parameters $\boldsymbol{\alpha} = (\alpha_1, \alpha_2, \alpha_3, \alpha_4)^T$ with total non-informative evidence mass $C = \sum_{k=1}^4 a_k \cdot W$ (where $W=4.0$ is the non-informative prior weight), the bijective mapping between evidence parameters $\boldsymbol{\alpha}$ and opinion $\omega = (\mathbf{b}, u, \mathbf{a})$ is given by:
\begin{equation}
b_k = \frac{\alpha_k - C \cdot a_k}{C + \sum_{m=1}^4 (\alpha_m - C \cdot a_m)}, \qquad u = \frac{C}{C + \sum_{m=1}^4 (\alpha_m - C \cdot a_m)}
\label{eq:dirichlet_bijection}
\end{equation}

The expected (projected) probability $\mathbf{P}(\theta_k)$ for each state $\theta_k \in \mathbb{X}$ is computed as:
\begin{equation}
\mathbf{P}(\theta_k) = b_k + a_k \cdot u
\label{eq:projected_prob}
\end{equation}

\subsection{Partial-Observable Projection}
When a sensory modality is missing or corrupted (e.g., optical satellite cloud occlusion or missing rain gauge observations), Kaplan's partial-observable projection rule \cite{kaplan2015subjective} projects evidence onto the unobserved dimensions:
\begin{equation}
\alpha_k^{\text{proj}} = C \cdot a_k \implies u \to 1.0, \quad b_k \to 0.0
\label{eq:kaplan_projection}
\end{equation}

\begin{IEEEproof}[Monotonicity Property]
Let $n_{\text{missing}} \in \{0, 1, 2, 3, 4\}$ denote the number of unobserved agent modalities. Under equation~(\ref{eq:kaplan_projection}), as $n_{\text{missing}}$ increases, the total evidence mass $\sum r_k$ decreases monotonically toward zero, forcing $u = \frac{C}{C + \sum r_k}$ to grow strictly monotonically toward $1.0$.
\end{IEEEproof}

\subsection{Multi-Source Fusion: WBF and CCF}
To combine opinions from $N \ge 2$ specialist agents, the coordinator computes fused opinion $\omega_{A,B} = \omega_A \oplus \omega_B$.

For Weighted Belief Fusion (WBF) weighted by agent credibility $\boldsymbol{\gamma} = (\gamma_A, \gamma_B)^T$ \cite{heijden2018distributed}:
\begin{equation}
b_k^{\text{WBF}} = \frac{\gamma_A b_k^A u_B + \gamma_B b_k^B u_A}{\gamma_A u_B + \gamma_B u_A - \gamma_A \gamma_B u_A u_B}, \quad u^{\text{WBF}} = \frac{u_A u_B}{\gamma_A u_B + \gamma_B u_A - \gamma_A \gamma_B u_A u_B}
\label{eq:wbf}
\end{equation}

When agents exhibit high mutual agreement, Consensus \& Compromise Fusion (CCF) is selected to redistribute belief masses proportionally without over-penalizing shared certainty \cite{heijden2018distributed}.

\subsection{Specialist-Credibility Update}
Each agent's dynamic credibility reputation $\gamma_i \in [\gamma_{\text{min}}, 1.0]$ (with $\gamma_{\text{min}} = 0.1$) updates at step $t+1$ based on its Brier score accuracy $\text{BS}_i$ against historical ground-truth state vector $\mathbf{y} \in \{0, 1\}^4$ \cite{wang2007formal}:
\begin{equation}
\text{BS}_i = \frac{1}{4} \sum_{k=1}^4 \left( \mathbf{P}_{A_i}(\theta_k) - y_k \right)^2
\label{eq:brier_score}
\end{equation}
\begin{equation}
\gamma_i^{(t+1)} = \text{clip}\left( \gamma_i^{(t)} + \eta \cdot (1 - 2 \cdot \text{BS}_i), \, \gamma_{\text{min}}, \, 1.0 \right)
\label{eq:credibility_update}
\end{equation}
where $\eta = 0.05$ is the learning rate.

\subsection{Hybrid-Head Composite Feature Space}
The final prediction head receives a 27-dimensional composite feature vector $\mathbf{F} \in \mathbb{R}^{27}$:
\begin{equation}
\mathbf{F} = \Big[ \underbrace{b_1, b_2, b_3, b_4, u, \text{max\_js}}_{\text{Fused Opinion (6)}} \, \Big| \, \underbrace{\gamma_1, \dots, \gamma_5}_{\text{Agent Credibility (5)}} \, \Big| \, \underbrace{x_1^{\text{raw}}, \dots, x_{16}^{\text{raw}}}_{\text{Raw Multimodal Features (16)}} \Big]
\label{eq:hybrid_feature_vector}
\end{equation}

Concatenating raw features alongside the SL opinion tensor allows gradient-boosted trees (LightGBM) to fit non-linear cross-modal interactions while leveraging calibrated epistemic uncertainty $u$ as a direct feature.
